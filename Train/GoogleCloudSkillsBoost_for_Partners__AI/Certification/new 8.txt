# Review the schema in func_arguments below, then observe how it is 
# passed to the parameters object of the FunctionDeclaration:
func_arguments = {
    # This argument is just a simple STRING
    "item_id": {
        "type": "STRING",
        "description": "Item's ID",
     },
    # This argument is an array of one or multiple STRINGs
    "item_color": {
        "description": "An array of colors of interest for an item.",
        "type": "ARRAY",
        "items": {
            "type": "STRING",
         },
    },
    # This argument is a STRING, but must match
    # one of the provided enum values to be valid
    "item_size": {
        "type": "STRING",

        "enum": ["XS", "S", "M", "L", "XL", "XXL"]
     },

########## Write a handler function for your model’s function call requests   for any AI SDK
def handle_response(response):

    # If the model has requested multiple function calls 
    # you will need to provide a response to each request.
    # This list will store those responses.  
    parts_for_inner_response = []

    # Iterate through the parts of Gemini's response
    for part in response.candidates[0].content.parts:
        
        # If the current part of the response is text, 
        # print it (or otherwise process it as you would
        # like to process a content response:
        if part.text:
            print("\n" + part.text)

        # If the current part of the response is a 
        # function_call, use its name to determine
        # which function to run
        if part.function_call:
            function_call = part.function_call

            if function_call.name == "check_inventory":

                # The names of the fields available within
                # function_call.args are those defined
                # by your function declaration.
                id = function_call.args["item_id"]
                colors = function_call.args["item_id"]
                size = function_call.args["item_size"]
                
                # Call the requested function
                inv_result = check_inv(id,
                                colors,
                                size)

                # Add the returned value to your list
                # of function results to provide the model.
                parts_for_inner_response.append(

                    # This is how you formally structure
                    # the response
                    Part.from_function_response(
                        name="check_inventory",
                        response={
                            "content": inv_result,
                        },
                    )
                )

                # If your model had another function
                # available, you could check for it here.
                if function_call.name == "another_function":
                    pass
                    
    # After each part has been checked and any function
    # results prepared, send them back to the model
    # for another response, and handle it recursively
    # (meaning call the same function again from within 
    # itself). This allows processing sequential function
    # calls.
    if parts_for_inner_response:
        inner_response = chat.send_message(
                             parts_for_inner_response
                         )
        handle_response(inner_response)


handle_response(response)

from IPython.display import Markdown, HTML

import vertexai
from vertexai.generative_models import (
    GenerationConfig,
    GenerativeModel,
    Tool,
    grounding,
)

PROJECT_ID = "YOUR-PROJECT-ID"
# To find your Data Store ID, navigate to 
# Agent Builder > Data Stores and select the 
# name of your data store from the list. 
# The data store ID will be displayed along 
# with other metadata about your data store.
data_store_id = "YOUR-DATA-STORE-ID"

vertexai.init(project=PROJECT_ID, location="us-central1")

model = GenerativeModel("gemini-1.5-flash-001")

tool = Tool.from_retrieval(
    grounding.Retrieval(
        grounding.VertexAISearch(
            datastore=data_store_id,
            project=PROJECT_ID,
            location="global",
        )
    )
)

prompt = "What are the differences between the Nest Thermostat and Nest Learning Thermostat?"

response = model.generate_content(
    prompt,
    tools=[tool],
    generation_config=GenerationConfig(
        temperature=0.0,
    ),
)

Markdown(response.text)
